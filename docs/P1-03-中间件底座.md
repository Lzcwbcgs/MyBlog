新建 `internal/middleware/base.go`：

```go
package middleware

import (
	"errors"
	g "gin-blog/internal/global"
	"gin-blog/internal/handle"
	"log/slog"
	"net"
	"net/http"
	"net/http/httputil"
	"net/url"
	"os"
	"runtime/debug"
	"strings"
	"time"

	"github.com/gin-contrib/cors"
	"github.com/gin-contrib/sessions"
	"github.com/gin-contrib/sessions/cookie"
	"github.com/gin-contrib/sessions/memstore"
	"github.com/gin-gonic/gin"
	"github.com/redis/go-redis/v9"
	"gorm.io/gorm"
)

// WithRedisDB 将 redis.Client 注入到 gin.Context
// handler 中通过 c.MustGet(g.CTX_RDB).(*redis.Client) 来使用
func WithRedisDB(rdb *redis.Client) gin.HandlerFunc {
	return func(ctx *gin.Context) {
		ctx.Set(g.CTX_RDB, rdb)
		ctx.Next()
	}
}

// WithGormDB 将 gorm.DB 注入到 gin.Context
// handler 中通过 c.MustGet(g.CTX_DB).(*gorm.DB) 来使用
func WithGormDB(db *gorm.DB) gin.HandlerFunc {
	return func(ctx *gin.Context) {
		ctx.Set(g.CTX_DB, db)
		ctx.Next()
	}
}

// CORS 跨域请求
/*
不能用 AllowOrigins: ["*"], 它会让响应头固定为 *,
而带凭证(AllowCredentials)的跨域请求在 * 下会被浏览器拒绝,
前后端分域名部署时 session cookie 就带不过去。所以用 AllowOriginFunc 回显具体 Origin。

但原来这个函数恒 return true, 「全放行 + 回显 Origin + 允许凭证」在安全上等同于任意站点
都能带着访客的登录态调评论、留言、改资料。改成白名单:
来源写在 config.yml 的 server.allowed-origins, 留空时只放行本机与内网(开发和内网自用场景)。
*/
func CORS() gin.HandlerFunc {
	allowed := g.Conf.AllowedOrigins()
	if len(allowed) == 0 {
		slog.Warn("未配置 server.allowed-origins, 跨域只放行本机与内网来源")
	}

	return cors.New(cors.Config{
		AllowMethods:     []string{"PUT", "POST", "GET", "DELETE", "OPTIONS", "PATCH"},
		AllowHeaders:     []string{"Origin", "Authorization", "Content-Type", "X-Requested-With"},
		ExposeHeaders:    []string{"Content-Type"},
		AllowCredentials: true,
		AllowOriginFunc: func(origin string) bool {
			return originAllowed(origin, allowed)
		},
		MaxAge: 24 * time.Hour,
	})
}

func originAllowed(origin string, allowed []string) bool {
	if origin == "" {
		return false
	}
	if len(allowed) > 0 {
		for _, item := range allowed {
			if strings.EqualFold(strings.TrimRight(item, "/"), strings.TrimRight(origin, "/")) {
				return true
			}
		}
		return false
	}
	return isLocalOrigin(origin)
}

// 本机 / 内网来源: localhost、127.x、::1 与私有网段。公网站点一律拦掉
func isLocalOrigin(origin string) bool {
	u, err := url.Parse(origin)
	if err != nil {
		return false
	}
	host := u.Hostname()
	if host == "localhost" || host == "::1" {
		return true
	}
	ip := net.ParseIP(host)
	if ip == nil {
		return false
	}
	return ip.IsLoopback() || ip.IsPrivate() || ip.IsLinkLocalUnicast()
}

// WithCookieStore 基于 cookie 的 session
func WithCookieStore(name, secret string) gin.HandlerFunc {
	store := cookie.NewStore([]byte(secret))
	store.Options(sessionOptions())
	return sessions.Sessions(name, store)
}

// WithMemStore 基于内存的 session
func WithMemStore(name, secret string) gin.HandlerFunc {
	store := memstore.NewStore([]byte(secret))
	store.Options(sessionOptions())
	return sessions.Sessions(name, store)
}

/*
session cookie 属性

原来只设了 Path 与 MaxAge: 没有 HttpOnly 意味着任意 XSS 脚本能读走会话,
没有 SameSite 则第三方站点发出的请求会自动带上它。
Secure 走配置(session.secure): 本地 http 调试时置 true 浏览器会直接丢掉 cookie。
*/
func sessionOptions() sessions.Options {
	return sessions.Options{
		Path:     "/",
		MaxAge:   600,
		HttpOnly: true,
		Secure:   g.Conf.SessionSecure(),
		SameSite: http.SameSiteLaxMode,
	}
}

// Logger 日志记录
func Logger() gin.HandlerFunc {
	return func(c *gin.Context) {
		start := time.Now()
		c.Next()
		cost := time.Since(start)

		slog.Info("[GIN]",
			slog.String("path", c.Request.URL.Path),
			slog.String("query", c.Request.URL.RawQuery),
			slog.Int("status", c.Writer.Status()),
			slog.String("method", c.Request.Method),
			slog.String("ip", c.ClientIP()),
			slog.Int("size", c.Writer.Size()),
			slog.Duration("cost", cost),
			// slog.String("body", c.Request.PostForm.Encode()),
			// slog.String("user-agent", c.Request.UserAgent()),
			// slog.String("errors", c.Errors.ByType(gin.ErrorTypePrivate).String()),
		)
	}
}

// Recovery 恢复中间件
func Recovery(stack bool) gin.HandlerFunc {
	return func(c *gin.Context) {
		defer func() {
			if err := recover(); err != nil {
				// Check for a broken connection, as it is not really a
				// condition that warrants a panic stack trace.
				var brokenPipe bool
				if ne, ok := err.(*net.OpError); ok {
					var se *os.SyscallError
					if errors.As(ne, &se) {
						seStr := strings.ToLower(se.Error())
						if strings.Contains(seStr, "broken pipe") ||
							strings.Contains(seStr, "connection reset by peer") {
							brokenPipe = true
						}
					}
				}

				// 发生 panic, 返回错误信息
				handle.ReturnHttpResponse(c, http.StatusInternalServerError, g.FAIL, g.GetMsg(g.FAIL), err)

				httpRequest, _ := httputil.DumpRequest(c.Request, false)
				if brokenPipe {
					slog.Error(c.Request.URL.Path,
						slog.Any("error", err),
						slog.String("request", string(httpRequest)),
					)
					// If the connection is dead, we can't write a status to it.
					_ = c.Error(err.(error)) // errcheck
					c.Abort()
					return
				}

				if stack {
					slog.Error("[Recovery from panic]",
						slog.Any("error", err),
						slog.String("request", string(httpRequest)),
						slog.String("stack", string(debug.Stack())),
					)
				} else {
					slog.Error("[Recovery from panic]",
						slog.Any("error", err),
						slog.String("request", string(httpRequest)),
					)
				}
				c.AbortWithStatus(http.StatusInternalServerError)
			}
		}()
		c.Next()
	}
}
```

## 3.3 逐段讲解

### ① 依赖注入的「塞入端」—— 中间件工厂模式

```go
func WithGormDB(db *gorm.DB) gin.HandlerFunc {
	return func(ctx *gin.Context) {
		ctx.Set(g.CTX_DB, db)
		ctx.Next()
	}
}
```

**这个结构在 Go 的中间件里到处都是，必须看懂：**

```
WithGormDB(db)          ← 外层函数：启动时调用一次
    ↓ 返回
func(ctx *gin.Context)  ← 内层函数：每次请求调用一次
```

**为什么要写成两层？**

因为 `r.Use(...)` 需要的是一个 `gin.HandlerFunc`，也就是 `func(*gin.Context)`。但你还得**把 `db` 传进去**。

Go 不支持「带参数的回调」，所以用**闭包捕获**：

```go
db := ginblog.InitDatabase(conf)     // 启动时：一个 *gorm.DB
r.Use(middleware.WithGormDB(db))     // 传进去，被闭包【捕获】
```

**捕获之后会怎样？** 内层函数每次都引用到**同一个** `db` 变量。所以：

- **调用一次，服务终身复用同一个连接池** —— 这是对的，不应该每个请求新建连接
- 如果想每请求一个独立的 DB 会话（比如带事务），那就要在 `ctx` 里放别的东西

**`ctx.Next()` 不能漏。**

gin 的中间件是**链式**的：每个中间件决定要不要把控制权交给下一个。

```go
return func(ctx *gin.Context) {
	ctx.Set(g.CTX_DB, db)
	ctx.Next()          // ← 把控制权交给链上的下一个
}
```

如果漏了 `ctx.Next()`，**后面的所有中间件和 handler 都不会执行**。

现象很有迷惑性：请求会返回 `200` 但响应体是空的，或者直接卡住。因为 gin 没东西可返回，就用默认的空响应。

> **记住这个模板**：
> ```go
> func XxxMiddleware(依赖) gin.HandlerFunc {
>     return func(c *gin.Context) {
>         // 前置逻辑
>         c.Next()
>         // 后置逻辑（可选）
>     }
> }
> ```
> `WithGormDB` 没有后置逻辑，所以 `c.Next()` 是最后一行。`Logger` 和 `Recovery` 有后置逻辑，所以后面还有代码。

**为什么叫「注入」（Injection）？**

对比全局变量的写法：

```go
var DB *gorm.DB     // 全局变量
func GetArticle(c *gin.Context) {
	db := DB         // 直接用
}
```

全局变量的问题是**测试时换不掉**。而 Context 注入是「按请求」的：

```go
// 测试时，自己造一个中间件塞测试库
r.Use(func(c *gin.Context) {
	c.Set(g.CTX_DB, testDB)    // handler 代码一行不改
	c.Next()
})
```

每个测试用例有自己的 context，互不干扰 —— **并发跑测试也不会打架**。这就是上一节 `GetDB` 用 `MustGet` 的全部意义。

### ② CORS —— 原版修过的高危配置（本节重点）

先看原版注释，它记了两个**不同层次**的问题：

> **问题一**：不能用 `AllowOrigins: ["*"]`，它会让响应头固定为 `*`，而带凭证（`AllowCredentials`）的跨域请求在 `*` 下会被浏览器拒绝，前后端分域名部署时 session cookie 就带不过去。
>
> **问题二**：但原来这个函数恒 `return true`，「全放行 + 回显 Origin + 允许凭证」在安全上等同于任意站点都能带着访客的登录态调评论、留言、改资料。

**这两个问题要分开理解。**

#### 问题一：`*` 和凭证不能共存

浏览器有**同源策略**：`a.com` 的页面默认不能读 `b.com` 的接口响应。

CORS 是「服务端主动放宽」的机制：服务端在响应里加几个头，告诉浏览器「我允许这个来源读我的东西」。

关键头是 `Access-Control-Allow-Origin`。两个选项：

```
Access-Control-Allow-Origin: *                    ← 允许所有来源
Access-Control-Allow-Origin: https://a.com        ← 只允许这一个
```

**但有个额外规则**：如果请求**带凭证**（cookie、HTTP 认证），那 `Access-Control-Allow-Origin` **不能是 `*`**，必须是具体来源。

浏览器这么做是因为：`*` 意味着「所有网站都能读」，而携带 cookie 的请求代表「用户已登录的身份」。允许任意网站带着用户的登录态读取数据 = 灾难。

所以原版注释说：用 `*` 的话，前后端分域名部署时 **session cookie 带不过去**。这不是安全考虑，是**功能坏了**。

#### 问题二：回显 Origin 解决了问题一，但引入了一个更严重的洞

要支持凭证，必须回显具体来源。所以改成 `AllowOriginFunc`：

```go
AllowOriginFunc: func(origin string) bool {
	return originAllowed(origin, allowed)
}
```

`cors` 这个库会：**回调返回 `true` 时，把请求的 `Origin` 原样回显到响应头里。**

于是有人就写成：

```go
AllowOriginFunc: func(origin string) bool {
	return true      // ❌ 恒真
}
```

**这行看着人畜无害，实际等价于「放弃同源保护」。**

#### 攻击过程

假设用户 Alice 登录了你的博客（cookie 里存着 session）。

1. Alice 在同一个浏览器里打开了 `evil.com`
2. `evil.com` 的页面里有一段 JS：

```js
fetch('https://你的博客/api/front/comment', {
  method: 'POST',
  credentials: 'include',        // ← 带上 cookie
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ content: '...', topic_id: 1, type: 1 })
})
```

3. 浏览器发出请求，**带上了 Alice 的 cookie**
4. 服务端看到 `Origin: https://evil.com`，回调返回 `true`，于是响应头里写 `Access-Control-Allow-Origin: https://evil.com` + `Access-Control-Allow-Credentials: true`
5. 浏览器看到「服务端允许 evil.com 带凭证」，于是**把响应交给 `evil.com` 的 JS**

**结果：`evil.com` 能冒充 Alice 发评论、发留言、改资料，甚至读到她的通知列表。**

> 这类攻击是 **CSRF**（跨站请求伪造）的 CORS 变体。传统 CSRF 靠 `SameSite` cookie 防，但这个洞绕过了那层 —— 因为浏览器是**主动允许**的。

#### 修复：白名单 + 正确的判断方式

```go
func originAllowed(origin string, allowed []string) bool {
	if origin == "" {
		return false
	}
	if len(allowed) > 0 {
		for _, item := range allowed {
			if strings.EqualFold(strings.TrimRight(item, "/"), strings.TrimRight(origin, "/")) {
				return true
			}
		}
		return false
	}
	return isLocalOrigin(origin)
}
```

**三段逻辑：**

**第一段：空 Origin 直接拒绝。**

什么时候 `Origin` 是空的？**同源请求**（浏览器不发 `Origin`）、**非浏览器请求**（curl、Postman）。

这些请求本来就不受同源策略约束，CORS 对它们没有意义。返回 `false` 表示「不添加 CORS 头」—— 不影响请求本身，只是不给跨域授权。

**第二段：配了白名单就严格匹配。**

```go
if len(allowed) > 0 {
	for _, item := range allowed {
		if strings.EqualFold(strings.TrimRight(item, "/"), strings.TrimRight(origin, "/")) {
			return true
		}
	}
	return false        // ← 注意：匹配不上直接 return false，不再走下面的兜底
}
```

两个细节：

- `strings.EqualFold` 而不是 `==` —— **大小写不敏感**。域名大小写理论上不敏感，但有些客户端会写 `HTTPS://Blog.Example.com`，用 `==` 会漏掉
- `strings.TrimRight(x, "/")` —— 去掉尾部斜杠。配置里写 `https://blog.example.com/`（带斜杠）和浏览器发的 `https://blog.example.com`（不带）应该视为相同

**第三段：没配白名单，退到「只放行本机与内网」。**

这是**开发场景的兜底**。你本地跑 `localhost:8888` 的前端，`localhost:8765` 的后端 —— 没有这层，本地开发会一直跨域失败。

**注意 `CORS()` 里那句警告：**

```go
allowed := g.Conf.AllowedOrigins()
if len(allowed) == 0 {
	slog.Warn("未配置 server.allowed-origins, 跨域只放行本机与内网来源")
}
```

**没配白名单时打警告** —— 这是个好习惯：告诉部署的人「你现在是开发模式的宽松配置，上生产前要配上」。

#### `isLocalOrigin` 里的安全细节

```go
func isLocalOrigin(origin string) bool {
	u, err := url.Parse(origin)
	if err != nil {
		return false
	}
	host := u.Hostname()       // ← 提取主机名
	...
}
```

**必须 `url.Parse` 后再取 `Hostname()`，不能字符串前缀匹配。**

对比错误的写法：

```go
// ❌ 危险
if strings.HasPrefix(origin, "http://localhost") {
	return true
}
```

攻击者注册一个域名 `localhost.evil.com`，然后发：

```
Origin: http://localhost.evil.com
```

`strings.HasPrefix` 检查通过 —— 因为字符串确实以 `http://localhost` 开头。于是攻击者的网站拿到了凭证授权。

**正确做法是解析成 URL 结构，取出真正的主机名再比较：**

- `url.Parse("http://localhost.evil.com").Hostname()` → `"localhost.evil.com"`
- 和 `"localhost"` 不相等 → 拒绝 ✅

> **这是个通用的安全原则**：判断「这个字符串属于某类」时，先把它**解析成结构化数据**，再判断结构。字符串匹配永远有边角情况。
>
> 同样的道理适用于：判断邮箱域名、判断文件扩展名、判断 URL 路径。原版在 `isLocalOrigin` 里做对了。

**再看 `net.ParseIP` 那一段：**

```go
ip := net.ParseIP(host)
if ip == nil {
	return false          // ← 不是 IP 也不是 localhost → 拒绝
}
return ip.IsLoopback() || ip.IsPrivate() || ip.IsLinkLocalUnicast()
```

`net` 包提供的判断函数：

| 方法 | 含义 | 例子 |
|---|---|---|
| `IsLoopback()` | 回环地址 | `127.0.0.1`、`::1` |
| `IsPrivate()` | 私有网段 | `10.x`、`172.16-31.x`、`192.168.x` |
| `IsLinkLocalUnicast()` | 链路本地 | `169.254.x`、`fe80::` |

**这是用标准库而不是自己写正则/网段判断** —— 又省事又不会错。

#### CORS 其余配置

```go
AllowMethods:     []string{"PUT", "POST", "GET", "DELETE", "OPTIONS", "PATCH"},
AllowHeaders:     []string{"Origin", "Authorization", "Content-Type", "X-Requested-With"},
ExposeHeaders:    []string{"Content-Type"},
AllowCredentials: true,
MaxAge:           24 * time.Hour,
```

逐个说：

- **`AllowMethods`** —— 允许的 HTTP 方法。列表里必须有 `OPTIONS`，因为浏览器发跨域请求前会先发一个 `OPTIONS` 预检请求问「我能发 POST 吗」。如果 `OPTIONS` 不在允许列表里，预检就失败了
- **`AllowHeaders`** —— 允许请求携带的**自定义头**。`Authorization` 必须在（JWT token 在这里）。`Content-Type` 是 `application/json` 请求必需的。少一个前端就会报错，而且**错误信息很难懂**（浏览器只说「CORS 失败」，不告诉你是哪个头没允许）
- **`ExposeHeaders`** —— 允许 JS **读取**的响应头。默认情况下 JS 只能读几个「安全头」，其余的需要显式暴露
- **`AllowCredentials: true`** —— 允许携带 cookie。这一条是前面所有安全讨论的前提
- **`MaxAge: 24 * time.Hour`** —— 预检结果缓存 24 小时。浏览器在这次之后、24 小时内不再发预检请求，直接发正式请求。**这能显著减少往返次数** —— 否则每个跨域请求都要两次（OPTIONS + 真实请求）

### ③ session 的两个实现与安全属性

```go
func WithCookieStore(name, secret string) gin.HandlerFunc {
	store := cookie.NewStore([]byte(secret))
	store.Options(sessionOptions())
	return sessions.Sessions(name, store)
}

func WithMemStore(name, secret string) gin.HandlerFunc {
	store := memstore.NewStore([]byte(secret))
	store.Options(sessionOptions())
	return sessions.Sessions(name, store)
}
```

**两种存储的区别：**

| | cookie store | memstore |
|---|---|---|
| 数据存哪 | **客户端 cookie 里**（加密签名） | **服务端内存** |
| 容量 | 4KB 上限 | 受内存限制 |
| 多实例 | **天然共享**（数据在客户端） | 不共享，每个实例各存一份 |
| 重启 | 不丢 | 丢 |
| 带宽 | 每次请求都传输 | 只传一个 ID |

**原版 `main.go` 用的是 cookie store。** 因为这个项目要能多实例部署，而 memstore 在多实例下会出「请求打到实例 A 登录，打到实例 B 未登录」的诡异现象。

**注意 `[]byte(secret)`。** `cookie.NewStore` 的签名是 `NewStore(keyPairs ...[]byte)` —— 它要的是**密钥字节**，用于对 cookie 内容做签名（防篡改）。配置里 `Session.Salt` 就是这个 `secret`。

**这就是 P0 里 `CheckSecrets` 校验 `Session.Salt` 的原因** —— 盐公开了，攻击者就能伪造 session cookie 内容。

#### cookie 属性逐个说

```go
func sessionOptions() sessions.Options {
	return sessions.Options{
		Path:     "/",
		MaxAge:   600,
		HttpOnly: true,
		Secure:   g.Conf.SessionSecure(),
		SameSite: http.SameSiteLaxMode,
	}
}
```

**`HttpOnly: true`** —— 防 XSS 偷会话。

```javascript
document.cookie    // ← HttpOnly 的 cookie 在这里读不到
```

如果网站有个 XSS 漏洞（比如评论区没转义），攻击者的脚本能执行 `document.cookie` 把会话偷走。`HttpOnly` 让这条路走不通。

**注意**：它防的是「JS 读取」，不防「JS 发起带 cookie 的请求」。后者要靠 `SameSite`。

**`SameSite: http.SameSiteLaxMode`** —— 防 CSRF。

三个值：

| 值 | 行为 |
|---|---|
| `None` | 所有跨站请求都带 cookie（**必须同时设 `Secure`**） |
| `Lax` | **顶层导航**的跨站请求带（点链接跳转），其他跨站请求（表单 POST、fetch、iframe）不带 |
| `Strict` | 所有跨站请求都不带 |

`Lax` 是**现代推荐** —— 它让「从别的网站点链接过来」这种正常场景仍然保持登录态，但拦住了「别的网站偷偷发 POST」的 CSRF。

**`Secure: g.Conf.SessionSecure()`** —— 只走 HTTPS。

设了之后，浏览器只在 HTTPS 连接上发送这个 cookie。防中间人嗅探（HTTP 是明文的，同一 WiFi 下能被抓包）。

**为什么走配置而不是硬编码 `true`？** 原版注释说得很清楚：

> 本地 http 调试时置 true 浏览器会直接丢掉 cookie。

本地开发是 `http://localhost:8765`，没有 HTTPS。**如果硬编码 `Secure: true`，浏览器会拒绝存储这个 cookie** —— 表现为「登录了但立刻又未登录」，而且**没有任何报错**（浏览器静默丢弃）。

这是个非常难查的问题，所以做成配置项。

**`MaxAge: 600`** —— 600 秒 = 10 分钟。

看起来很短，但 **P3 的 `JWTAuth` 中间件每次请求都会刷新 session**：

```go
// middleware/auth.go（P3）
session := sessions.Default(c)
session.Set(g.CTX_USER_AUTH, claims.UserId)
session.Save()      // ← 每次请求都存，等于续期
```

所以实际的过期规则是「**连续 10 分钟没有任何请求才失效**」。这是个常见的设计：**滑动过期**（sliding expiration）。

**`Path: "/"`** —— 整个站点都发这个 cookie。如果填 `/api`，那 `/admin` 下的页面就收不到。

### ④ 自定义 `Logger` —— 为什么不用 gin 自带的

```go
func Logger() gin.HandlerFunc {
	return func(c *gin.Context) {
		start := time.Now()
		c.Next()
		cost := time.Since(start)

		slog.Info("[GIN]",
			slog.String("path", c.Request.URL.Path),
			...
			slog.Duration("cost", cost),
		)
	}
}
```

**结构：`c.Next()` 之前记开始时间，之后记结束时间。**

这是中间件做「计时」的标准手法 —— 因为 `c.Next()` 会**阻塞到后面所有 handler 执行完**才返回。

```
start := time.Now()      ←──┐
c.Next()                     │  这里执行了所有后续中间件 + handler
cost := time.Since(start) ←──┘
```

**`slog` 相比 gin 自带 `Logger()` 的好处：**

gin 自带的输出长这样（固定格式，改不了）：

```
[GIN] 2026/09/19 - 22:30:57 | 200 |  526.9µs | ::1 | GET "/api/ping"
```

自定义的输出是**结构化的 key-value**：

```
time=2026-09-19T22:30:57 level=INFO msg=[GIN] path=/api/ping status=200 method=GET ip=::1 size=31 cost=526.9µs
```

**差别在哪：**

- **能被解析**：`slog` 可以配成 JSON 格式（`config.yml` 里 `Log.Format: json`），日志采集系统（ELK、Loki）能直接字段化查询 —— 「找出所有 cost > 1s 的请求」变成一条查询语句
- **字段可控**：想加什么加什么，想删什么删什么
- **统一出口**：和项目里其他日志（`ReturnError` 里的 `slog.Error`）走同一套级别控制和格式

**注意被注释掉的三行：**

```go
// slog.String("body", c.Request.PostForm.Encode()),
// slog.String("user-agent", c.Request.UserAgent()),
// slog.String("errors", c.Errors.ByType(gin.ErrorTypePrivate).String()),
```

**原版为什么把它们注掉？** 日志噪音。

- `body` —— 请求体可能有几十 KB（比如文章内容），每条日志都记会把日志文件撑爆。**而且可能包含密码** —— 操作日志中间件（P4）会专门做脱敏，这里就不该再记一遍
- `user-agent` —— 大部分时候没用，需要时去操作日志里查
- `errors` —— 大部分请求没有 error，白白多一个空字段

> **这是日志设计的一个通用原则**：**高频日志只记必要字段**。请求日志每天可能几十万条，每条多 1KB 就是几百 MB。
>
> 需要详细信息时，用**采样**（比如只记 1% 的完整信息）或者**专门的操作日志**（P4 的 `operation_log` 表，只记写操作）。

### ⑤ 自定义 `Recovery` —— 客户端断连不是 bug

```go
func Recovery(stack bool) gin.HandlerFunc {
	return func(c *gin.Context) {
		defer func() {
			if err := recover(); err != nil {
				...
			}
		}()
		c.Next()
	}
}
```

**这是 Go 的 panic 恢复模板：**

```go
defer func() {
	if err := recover(); err != nil {
		// 处理 panic
	}
}()
```

`recover()` 只能在 `defer` 里生效，而且必须**直接调用**（不能包一层函数）。这是 Go 的规则。

**为什么要 `defer`？** 因为 `c.Next()` 里如果 panic 了，控制权会沿调用栈向上传播，`defer` 是唯一能拦截它的地方。

#### broken pipe 检测 —— 本段的核心

```go
var brokenPipe bool
if ne, ok := err.(*net.OpError); ok {
	var se *os.SyscallError
	if errors.As(ne, &se) {
		seStr := strings.ToLower(se.Error())
		if strings.Contains(seStr, "broken pipe") ||
			strings.Contains(seStr, "connection reset by peer") {
			brokenPipe = true
		}
	}
}
```

**要解决的问题**：用户在页面加载到一半时关掉标签页，或者刷新页面。

这时候：

1. 客户端 TCP 连接断了
2. 服务端还在处理请求，然后试图写响应
3. 写失败 → 返回一个错误
4. gin 把它变成 panic

**如果按普通 panic 处理会怎样？**

- 打印完整堆栈（几十行日志）
- 记成 Error 级别
- 触发告警

**但这根本不是 bug** —— 用户关页面是最正常不过的行为。真正的 bug 会被淹没在这些噪音里。

**所以原版做了类型检查：**

```go
err.(*net.OpError)          // 网络操作错误
errors.As(ne, &se)          // 里面是不是 *os.SyscallError
```

`errors.As` 是 Go 1.13+ 的错误链遍历 —— 它在错误链里查找指定类型的错误。这里用来「剥开」`net.OpError`，看里面的 `Err` 字段是不是 `os.SyscallError`。

判断到这一步之后，只能靠**字符串匹配**：

```go
strings.Contains(seStr, "broken pipe") ||
strings.Contains(seStr, "connection reset by peer")
```

> **这里为什么可以用字符串匹配，而 CORS 那里不行？**
>
> 因为**这里不涉及安全决策** —— 最坏情况是「误判成 broken pipe，少打了一条日志」。而 CORS 那里误判会导致「攻击者拿到凭证授权」。
>
> **判断标准的严格程度，取决于判错的代价。** 这是读这段代码时值得体会的一点。
>
> 而且操作系统层面的错误本来就是字符串 —— Go 标准库没有为这类错误提供类型化的判断方法。

**检测到 brokenPipe 之后的处理：**

```go
if brokenPipe {
	slog.Error(c.Request.URL.Path,
		slog.Any("error", err),
		slog.String("request", string(httpRequest)),
	)
	_ = c.Error(err.(error))    // 挂到 c.Errors 上，不写响应
	c.Abort()
	return
}
```

**关键点：不写响应。**

因为连接已经死了，写什么都是白写（还会再次失败）。所以只记日志 + `c.Abort()` 中止流程。

**注意 `err.(error)` 这个类型断言** —— `recover()` 返回的是 `any`，但 `c.Error()` 要求 `error` 类型。这里直接断言，如果 panic 的不是 error（比如 `panic("字符串")`），**会二次 panic**。

> 这是个**潜在的小缺陷** —— 应该写成 `if e, ok := err.(error); ok { c.Error(e) }`。但实践里 `panic` 的对象几乎总是 error，所以原版没管。
>
> **这类「理论上有问题但实践中不会触发」的地方，在真实项目里到处都是。** 能识别出它们，也是读代码的能力。

**`stack` 参数控制堆栈：**

```go
if stack {
	slog.Error("[Recovery from panic]", ..., slog.String("stack", string(debug.Stack())))
} else {
	slog.Error("[Recovery from panic]", ...)     // 不记堆栈
}
```

`main.go` 里传的是 `middleware.Recovery(true)` —— 记堆栈。

**什么时候会传 `false`？** 如果生产环境担心日志量（堆栈一条可能几 KB），可以关掉。但对这个项目的规模，开着更好 —— panic 本来就不该频繁发生。

**`debug.Stack()`** 返回当前 goroutine 的调用栈，是定位 panic 的关键信息。

#### 一个可以商量的地方

原版在 panic 时调了两次「写响应」：

```go
handle.ReturnHttpResponse(c, http.StatusInternalServerError, g.FAIL, g.GetMsg(g.FAIL), err)   // ← 写了 500
...
c.AbortWithStatus(http.StatusInternalServerError)                                              // ← 又写一次 500
```

`ReturnHttpResponse` 内部是 `c.JSON(500, ...)`，已经写完了。`AbortWithStatus` 又设了一次状态码。

**为什么没出问题？** 因为 gin 的 `responseWriter` 里对重复设状态码做了保护：

```go
func (w *responseWriter) WriteHeader(code int) {
	if code > 0 && w.status != code {
		if w.Written() {
			debugPrint("[WARNING] Headers were already written. ...")
		}
		w.status = code
	}
}
```

两次都是 `500`，所以连警告都不会有。**但这是冗余的** —— `AbortWithStatus` 在这里的唯一有效作用是 `Abort()`（中止后续 handler）。

写得更干净的话应该是：

```go
handle.ReturnHttpResponse(c, http.StatusInternalServerError, g.FAIL, g.GetMsg(g.FAIL), err)
c.Abort()          // 只需要中止，不需要再设状态码
```

> **我们按原版复刻，保留这段。** 但你要知道它是冗余的 —— 读别人的代码时能看出「这里多写了一句」，是个有用的能力。
